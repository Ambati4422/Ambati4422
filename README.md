def shade(h, f):
    r, g, b = int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16)
    if f >= 1: r, g, b = [int(c + (255 - c) * (f - 1)) for c in (r, g, b)]
    else:      r, g, b = [int(c * f) for c in (r, g, b)]
